+/**
+ * createDeckBar — DJ UI: Deck A | crossfader + focus | Deck B + library chip row
+ *
+ * @param {HTMLElement} rootEl  - container element (section#deckbar)
+ * @param {{
+ *   onAddFiles: (FileList) => void,
+ *   onLoadToFocused: (File) => void,
+ *   onFocus: (slot: 'a'|'b') => void,
+ *   onCrossfade: (value: number) => void,
+ *   onPlayToggle: (slot: 'a'|'b') => void,
+ * }} callbacks
+ * @returns {{
+ *   setDeck: (slot: 'a'|'b', {name: string, playing: boolean}) => void,
+ *   setFocus: (slot: 'a'|'b') => void,
+ *   setLibrary: (files: File[]) => void,
+ *   getCrossfade: () => number,
+ * }}
+ */
+export function createDeckBar(rootEl, {
+  onAddFiles,
+  onLoadToFocused,
+  onFocus,
+  onCrossfade,
+  onPlayToggle,
+}) {
+  // ── Build DOM ─────────────────────────────────────────────────────────────
+
+  rootEl.innerHTML = '';
+
+  // Top strip: Deck A | center controls | Deck B
+  const strip = el('div', 'db-strip');
+  rootEl.appendChild(strip);
+
+  // Deck A
+  const { panel: panelA, nameEl: nameA, playBtn: playA } = makeDeck('A', () => onPlayToggle('a'));
+  strip.appendChild(panelA);
+
+  // Center column: crossfader + focus
+  const center = el('div', 'db-center');
+  strip.appendChild(center);
+
+  const cfLabel = el('div', 'db-cf-label');
+  cfLabel.textContent = 'A ← crossfade → B';
+  center.appendChild(cfLabel);
+
+  const crossfader = document.createElement('input');
+  crossfader.type = 'range';
+  crossfader.min = '0';
+  crossfader.max = '1';
+  crossfader.step = '0.01';
+  crossfader.value = '0.5';
+  crossfader.className = 'db-crossfader';
+  crossfader.addEventListener('input', () => onCrossfade(parseFloat(crossfader.value)));
+  center.appendChild(crossfader);
+
+  const focusRow = el('div', 'db-focus-row');
+  center.appendChild(focusRow);
+
+  const focusLabel = el('span', 'db-focus-label');
+  focusLabel.textContent = 'Focus';
+  focusRow.appendChild(focusLabel);
+
+  const focusBtnA = document.createElement('button');
+  focusBtnA.textContent = 'A';
+  focusBtnA.className = 'db-focus-btn db-focus-a';
+  focusBtnA.addEventListener('click', () => onFocus('a'));
+  focusRow.appendChild(focusBtnA);
+
+  const focusBtnB = document.createElement('button');
+  focusBtnB.textContent = 'B';
+  focusBtnB.className = 'db-focus-btn db-focus-b';
+  focusBtnB.addEventListener('click', () => onFocus('b'));
+  focusRow.appendChild(focusBtnB);
+
+  // Deck B
+  const { panel: panelB, nameEl: nameB, playBtn: playB } = makeDeck('B', () => onPlayToggle('b'));
+  strip.appendChild(panelB);
+
+  // Library row
+  const libraryRow = el('div', 'db-library');
+  rootEl.appendChild(libraryRow);
+
+  const fileInput = document.createElement('input');
+  fileInput.type = 'file';
+  fileInput.accept = 'audio/*';
+  fileInput.multiple = true;
+  fileInput.id = 'deckLibraryFile';
+  fileInput.style.display = 'none';
+  fileInput.addEventListener('change', () => {
+    if (fileInput.files && fileInput.files.length > 0) {
+      onAddFiles(fileInput.files);
+      fileInput.value = ''; // reset so same file can be re-added
+    }
+  });
+  libraryRow.appendChild(fileInput);
+
+  const addLabel = document.createElement('label');
+  addLabel.htmlFor = 'deckLibraryFile';
+  addLabel.className = 'db-add-btn';
+  addLabel.textContent = '＋ Add tracks';
+  libraryRow.appendChild(addLabel);
+
+  const chipList = el('div', 'db-chip-list');
+  libraryRow.appendChild(chipList);
+
+  // ── Public API ────────────────────────────────────────────────────────────
+
+  function setDeck(slot, { name, playing }) {
+    if (slot === 'a') {
+      nameA.textContent = name || '— no track —';
+      playA.textContent = playing ? '⏸' : '▶';
+      panelA.classList.toggle('db-deck--playing', !!playing);
+    } else {
+      nameB.textContent = name || '— no track —';
+      playB.textContent = playing ? '⏸' : '▶';
+      panelB.classList.toggle('db-deck--playing', !!playing);
+    }
+  }
+
+  function setFocus(slot) {
+    focusBtnA.classList.toggle('db-focus-btn--on', slot === 'a');
+    focusBtnB.classList.toggle('db-focus-btn--on', slot === 'b');
+    panelA.classList.toggle('db-deck--focused', slot === 'a');
+    panelB.classList.toggle('db-deck--focused', slot === 'b');
+  }
+
+  function setLibrary(files) {
+    chipList.innerHTML = '';
+    for (const file of files) {
+      const chip = document.createElement('button');
+      chip.className = 'db-chip';
+      chip.textContent = file.name.replace(/\.[^.]+$/, ''); // strip extension
+      chip.title = file.name;
+      chip.addEventListener('click', () => onLoadToFocused(file));
+      chipList.appendChild(chip);
+    }
+  }
+
+  function getCrossfade() {
+    return parseFloat(crossfader.value);
+  }
+
+  // Set initial focus to A
+  setFocus('a');
+  setDeck('a', { name: '', playing: false });
+  setDeck('b', { name: '', playing: false });
+
+  return { setDeck, setFocus, setLibrary, getCrossfade };
+}
+
+// ── Helpers ──────────────────────────────────────────────────────────────────
+
+function el(tag, className) {
+  const node = document.createElement(tag);
+  if (className) node.className = className;
+  return node;
+}
+
+function makeDeck(label, onPlayToggle) {
+  const panel = el('div', 'db-deck');
+  panel.dataset.slot = label.toLowerCase();
+
+  const header = el('div', 'db-deck-header');
+  const slotLabel = el('span', 'db-deck-slot');
+  slotLabel.textContent = `Deck ${label}`;
+  header.appendChild(slotLabel);
+  panel.appendChild(header);
+
+  const nameEl = el('div', 'db-deck-name');
+  nameEl.textContent = '— no track —';
+  panel.appendChild(nameEl);
+
+  const playBtn = document.createElement('button');
+  playBtn.className = 'db-play-btn';
+  playBtn.textContent = '▶';
+  playBtn.addEventListener('click', onPlayToggle);
+  panel.appendChild(playBtn);
+
+  return { panel, nameEl, playBtn };
+}
